297    async def start_measure(self, ch: Optional[int] = None, on: Optional[int] = 1) -> bytes:
298        try:
299            if ch:
300                if on:
301                    STATE_MEASURE = self.MPP_START_MEASURE.copy()
302                    STATE_MEASURE[0] = ch & 0xFF << 8 | STATE_MEASURE[0] & 0xFFFF
303                else:
304                    STATE_MEASURE = self.MPP_STOP_MEASURE.copy()
305                    STATE_MEASURE[0] = ch & 0xFF << 8 | STATE_MEASURE[0] & 0xFFFF
306                result: ModbusResponse = await self.client.write_registers(self.REG_MPP_CTRL, 
307                                                                            STATE_MEASURE,
308                                                                            slave=self.MPP_ID)
309                await log_s(self.mw.send_handler.mess)
310                await self.client.read_holding_registers(self.REG_MPP_CTRL, 1, self.MPP_ID)
311                await log_s(self.mw.send_handler.mess)
312                if on:
313                    await self.issue_waveform()
314                    await log_s(self.mw.send_handler.mess)
315                    await self.client.read_holding_registers(self.REG_MPP_CTRL, 1, self.MPP_ID)
316            else:
317                if on:
318                    STATE_MEASURE = self.MPP_START_MEASURE
319                else:
320                    STATE_MEASURE = self.MPP_STOP_MEASURE
321                result: ModbusResponse = await self.client.write_registers(self.REG_MPP_CTRL, 
322                                                                            STATE_MEASURE,
323                                                                            slave=self.MPP_ID)
324                await log_s(self.mw.send_handler.mess)
325                await self.client.read_holding_registers(self.REG_MPP_CTRL, 1, self.MPP_ID)
326                await log_s(self.mw.send_handler.mess)
327                if on:
328                    await self.client.write_registers(self.REG_MPP_CTRL, 
329                                                                            0x0009,
330                                                                            slave=self.MPP_ID) # выдать waveform
331                    await log_s(self.mw.send_handler.mess)
332                    await self.client.read_holding_registers(self.REG_MPP_CTRL, 1, self.MPP_ID)
333                    await log_s(self.mw.send_handler.mess)
334
335            return result.encode()
336        except Exception as e:
337            self.logger.error(e)
338            self.logger.debug('МПП не отвечает')
339            return b'-1'
