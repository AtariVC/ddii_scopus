138    async def get_client(self) -> None:
139        """Функция перехватывает client и переподключается к нему
140        """

158                        self.radioButton_cm.setChecked(False)
159                        self.radioButton_cm.setEnabled(False)
160                if self.w_ser_dialog.status_CM == 0 and self.w_ser_dialog.status_MPP == 0:
161                    self.radioButton_cm.setChecked(True)
162                    self.radioButton_cm.setEnabled(False)
163                    self.radioButton_mpp.setEnabled(False)

166            pass

174            answ_lvl: bytes = await self.mpp_cmd.get_level()

176
177            self.lineEdit_lvl_0_1.setText(str(tel_dict_lvl["01_hh_l"]))
178
179            for i, (key, val) in enumerate(self.le_obj.items()):
180                if 0 < i < 9:
181                    val.setText(list(tel_dict.values())[i-1])
182
183        except Exception as e:
184            self.logger.error(e)

241    @qasync.asyncSlot()
242    async def check_writed_cfg(self, data: list[int], device: str) -> bool:
243        """Поверяет записалась ли в память конфигурация
244        Args:
245            data (list[int]): отправленные данные концигурации
246            device (str): 
247            - "cm"
248            - "mpp"
249        Returns:
250            bool: Статус проверки
251        """
252        # await self.cm_cmd.set_mode(self.SILENT_MODE)
253        # await asyncio.sleep(0.5)
254        try:
255            if device == "mpp":
256                cheack_lvl: bytes = await self.mpp_cmd.get_level()
257                cheack_hh: bytes = await self.mpp_cmd.get_hh()
258                d_cheack_lvl: dict[str, str] =  await self.parser.pars_mpp_lvl(cheack_lvl)
259                d_cheack_hh: dict[str, str] = await self.parser.pars_mpp_hh(cheack_hh)
260                if list(map(int, d_cheack_hh.values())) == data[1:9] and \
261                        list(map(int, d_cheack_lvl.values())) == data[:1]:
262                    return True
263                else:
264                    return False
265        except Exception as e:
266            self.logger.error(e)
267
268        try:
269            if device == "cm": # для цм не работает из-за точности float
270                cheack_cfg_ddii: bytes = await self.cm_cmd.get_cfg_ddii()
271                d_cheack_cfg_ddii:list[int] =  [int.from_bytes(cheack_cfg_ddii[i*2:i*2+2], "little") for i in range(2, 24)]
272                if d_cheack_cfg_ddii == data[1:]:
273                    return True
274                else:
275                    return False
276        except Exception as e:
277            self.logger.error(e)
278        # await asyncio.sleep(0.5)
279        # await self.cm_cmd.set_mode(self.COMBAT_MODE)
280        return False
